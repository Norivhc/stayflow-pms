-- ============================================================================
-- ESQUEMA DE BASE DE DATOS - STAYFLOW PMS (POSTGRESQL / SUPABASE)
-- ============================================================================

-- 1. Tabla maestra de huéspedes (CRM)
CREATE TABLE huespedes (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL,
    nombre TEXT NOT NULL,
    telefono TEXT,
    dni TEXT,
    procedencia TEXT
);

-- 2. Tabla transaccional de reservas, check-in y liquidación
CREATE TABLE reservas (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL,
    huesped_id BIGINT REFERENCES huespedes(id) ON DELETE SET NULL,
    huesped_nombre TEXT NOT NULL,
    telefono TEXT,
    dni_titular TEXT,
    procedencia TEXT,
    notas TEXT,
    departamento TEXT NOT NULL,
    checkin DATE NOT NULL,
    checkout DATE NOT NULL,
    hora_checkout TEXT DEFAULT '10:00',
    extra_horas NUMERIC NOT NULL DEFAULT 0,
    personas INTEGER NOT NULL DEFAULT 1,
    noches INTEGER NOT NULL,
    precio_noche NUMERIC NOT NULL,
    total NUMERIC NOT NULL,
    senia NUMERIC NOT NULL DEFAULT 0,
    comision NUMERIC NOT NULL,
    comision_cobrada NUMERIC NOT NULL DEFAULT 0,
    estado_pago TEXT DEFAULT 'Señado',
    acompanantes JSONB DEFAULT '[]'::jsonb
);

-- 3. Políticas de seguridad a nivel de fila (RLS) y permisos API
ALTER TABLE huespedes ENABLE ROW LEVEL SECURITY;
ALTER TABLE reservas ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Acceso total huespedes" ON huespedes FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Acceso total reservas" ON reservas FOR ALL USING (true) WITH CHECK (true);

GRANT ALL ON TABLE huespedes TO anon, authenticated;
GRANT ALL ON TABLE reservas TO anon, authenticated;
GRANT USAGE, SELECT ON ALL SEQUENCES IN SCHEMA public TO anon, authenticated;
